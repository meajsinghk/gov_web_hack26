EPF Corpus Accumulation CalculatorStatutory Formula: Compounded annually at the gazetted interest rate $r = 8.25\%$.Monthly Contribution:$$\text{Employee Share} = 12\% \times \text{Basic Wage}$$$$\text{Employer EPF Share} = 3.67\% \times \text{Basic Wage}$$$$\text{Total Monthly Addition} = 15.67\% \times \text{Basic Wage}$$Annual Compound Growth:$$A_n = (A_{n-1} + 12 \times \text{Monthly Addition}) \times \left(1 + \frac{r}{100}\right)$$Includes projected annual wage increment parameter ($5\% - 10\%$).2. EPS 95 Monthly Pension Entitlement CalculatorStatutory Formula (Employees' Pension Scheme, 1995):$$\text{Monthly Pension} = \frac{\text{Pensionable Service} \times \text{Pensionable Salary}}{70}$$Statutory Rules & Edge Constraints:Pensionable Salary: Average monthly wage of the last 60 months, capped at ₹15,000 (or actual average salary if the higher pension joint option is active).Pensionable Service Bonus: If total service $\ge 20\text{ years}$, add $2\text{ bonus years}$ automatically.Early Pension Penalty: If retired between ages 50 and 57, reduce pension by $4\%$ for each year under 58.Deferred Pension Bonus: If deferred past age 58 up to 60, increase pension by $4\%$ per year.Minimum Monthly Pension: Floor guaranteed at ₹1,000/month.3. EDLI Group Term Life Insurance Benefit EstimatorStatutory Formula (Employees' Deposit-Linked Insurance Scheme, 1976):$$\text{EDLI Benefit} = (35 \times \text{Average Monthly Wages of Last 12 Months}) + \text{Bonus Amount}$$Statutory Rules & Edge Constraints:Wage Ceiling: Capped at ₹15,000 per month (yielding maximum base benefit of $35 \times 15,000 = ₹5,25,000$).Bonus Component: ₹1,75,000 added automatically.Maximum Cap: ₹7,00,000 (₹7 Lakh).Minimum Assurance Floor: ₹2,50,000 for members with continuous service $\ge 12\text{ months}$; ₹50,000 for death occurring within the first year of service.4. // Edge Pre-Flight Verification & Jaro-Winkler String Distance Engine
// ==========================================================================
class PreFlightEngine {
  /**
   * Evaluates Jaro-Winkler string similarity distance between two strings
   */
  static jaroWinkler(s1, s2) {
    s1 = s1.trim().toUpperCase();
    s2 = s2.trim().toUpperCase();
    if (s1 === s2) return 1.0;

    const len1 = s1.length;
    const len2 = s2.length;
    const matchDistance = Math.floor(Math.max(len1, len2) / 2) - 1;

    const s1Matches = new Array(len1).fill(false);
    const s2Matches = new Array(len2).fill(false);

    let matches = 0;
    let transpositions = 0;

    for (let i = 0; i < len1; i++) {
      const start = Math.max(0, i - matchDistance);
      const end = Math.min(i + matchDistance + 1, len2);
      for (let j = start; j < end; j++) {
        if (s2Matches[j] || s1[i] !== s2[j]) continue;
        s1Matches[i] = true;
        s2Matches[j] = true;
        matches++;
        break;
      }
    }

    if (matches === 0) return 0.0;

    let k = 0;
    for (let i = 0; i < len1; i++) {
      if (!s1Matches[i]) continue;
      while (!s2Matches[k]) k++;
      if (s1[i] !== s2[k]) transpositions++;
      k++;
    }

    const m = matches;
    const dj = (m / len1 + m / len2 + (m - transpositions / 2) / m) / 3;
    let l = 0;
    while (l < 4 && s1[l] === s2[l]) l++;

    return dj + l * 0.1 * (1 - dj);
  }

  /**
   * Full Pre-Flight Gate Validation
   */
  static executePreFlightCheck(citizen, requestedForm, requestedAmount) {
    const results = {
      passed: true,
      checks: []
    };

    // 1. Identity Name Congruence (Aadhaar vs Bank)
    const nameScore = this.jaroWinkler(citizen.aadhaar.full_name, citizen.bank.holder_name);
    if (nameScore >= 0.90) {
      results.checks.push({ name: "Identity & Bank Name Match", status: "PASS", detail: `Score: ${(nameScore * 100).toFixed(1)}%` });
    } else if (nameScore >= 0.75) {
      results.checks.push({ name: "Identity Reconciliation Needed", status: "WARN", detail: "Minor variation detected. OTP verification required." });
    } else {
      results.passed = false;
      results.checks.push({ name: "Identity Mismatch", status: "FAIL", detail: "Bank name does not match Aadhaar registry." });
    }

    // 2. NPCI Bank Mapping
    if (citizen.bank.npci_mapped && citizen.bank.penny_drop_valid) {
      results.checks.push({ name: "NPCI Bank Seeding", status: "PASS", detail: "Direct Account Disbursal Ready" });
    } else {
      results.passed = false;
      results.checks.push({ name: "NPCI Bank Seeding", status: "FAIL", detail: "Bank not seeded with NPCI." });
    }

    // 3. Statutory Scheme Verification
    if (requestedForm === "FORM_19" && !citizen.service.doe) {
      results.passed = false;
      results.checks.push({ name: "Date of Exit", status: "FAIL", detail: "Active employment detected. Form 19 requires marked exit." });
    } else {
      results.checks.push({ name: "Statutory Eligibility", status: "PASS", detail: "Valid scheme parameters verified." });
    }

    // 4. Balance Sufficiency Check
    if (requestedAmount > citizen.balances.total_withdrawable) {
      results.passed = false;
      results.checks.push({ name: "Balance Sufficiency", status: "FAIL", detail: "Requested amount exceeds withdrawable corpus." });
    } else {
      results.checks.push({ name: "Corpus Sufficiency", status: "PASS", detail: "Funds available for settlement." });
    }

    return results;
  }
}

// Statutory Calculators Suite (EPF, EPS 95, EDLI & Retirement Score)
// ==========================================================================
class StatutoryCalculators {
  /**
   * 1. EPF Corpus Accumulation Calculator (8.25% compound interest)
   */
  static calculateEPF(basicSalary, currentAge, retirementAge = 58, currentCorpus = 0, annualIncrementPct = 5) {
    let balance = currentCorpus;
    let salary = basicSalary;
    const interestRate = 0.0825;
    const years = Math.max(0, retirementAge - currentAge);
    const yearlyBreakdown = [];

    for (let yr = 1; yr <= years; yr++) {
      const monthlyContribution = salary * 0.1567; // 12% Emp + 3.67% Employer EPF
      const annualAdditions = monthlyContribution * 12;
      balance = (balance + annualAdditions) * (1 + interestRate);
      salary = salary * (1 + annualIncrementPct / 100);

      yearlyBreakdown.push({
        year: yr,
        age: currentAge + yr,
        corpus: Math.round(balance),
        monthlyWage: Math.round(salary)
      });
    }

    return {
      finalCorpus: Math.round(balance),
      totalYears: years,
      yearlyBreakdown
    };
  }

  /**
   * 2. EPS 95 Monthly Pension Calculator
   */
  static calculateEPS(pensionableSalary, serviceYears, retirementAge = 58) {
    // Statutory ceiling capped at 15000 unless higher wage option is exercised
    const cappedSalary = Math.min(pensionableSalary, 15000);
    let effectiveService = serviceYears;

    // 2-year service bonus if completed tenure >= 20 years
    if (effectiveService >= 20) {
      effectiveService += 2;
    }

    let monthlyPension = (effectiveService * cappedSalary) / 70;

    // Early retirement adjustment (between 50 and 57 years: 4% reduction per year)
    if (retirementAge < 58 && retirementAge >= 50) {
      const reductionYears = 58 - retirementAge;
      monthlyPension = monthlyPension * (1 - (reductionYears * 0.04));
    }

    // Minimum statutory pension floor = Rs 1000
    monthlyPension = Math.max(1000, monthlyPension);

    return {
      monthlyPension: Math.round(monthlyPension),
      effectiveServiceYears: effectiveService,
      wageConsidered: cappedSalary
    };
  }

  /**
   * 3. EDLI Insurance Assurance Benefit Estimator
   */
  static calculateEDLI(average12MonthWage) {
    // Wage capped at 15000
    const wage = Math.min(average12MonthWage, 15000);
    const baseBenefit = 35 * wage;
    const bonus = 175000;
    let totalBenefit = baseBenefit + bonus;

    // Enforce Statutory Floor (Rs 2.5 Lakh) and Maximum Cap (Rs 7 Lakh)
    totalBenefit = Math.max(250000, Math.min(700000, totalBenefit));

    return {
      edliCoverAmount: totalBenefit,
      baseBenefit,
      bonusAdded: bonus,
      wageCapped: wage
    };
  }

  /**
   * 4. Retirement Readiness Score (0 to 100)
   */
  static calculateReadinessScore(currentCorpus, monthlyWage, currentAge, projectedPension) {
    const targetCorpus = monthlyWage * 12 * 15; // 15x annual wage heuristic
    const currentWeight = (currentCorpus / targetCorpus) * 60;
    const pensionWeight = (projectedPension / (monthlyWage * 0.5)) * 40;
    const score = Math.min(100, Math.round(currentWeight + pensionWeight));

    let tier = "Needs Attention";
    if (score >= 80) tier = "Excellent - On Track";
    else if (score >= 50) tier = "Moderate - Steady Progress";

    return { score, tier, targetCorpus };
  }
}
